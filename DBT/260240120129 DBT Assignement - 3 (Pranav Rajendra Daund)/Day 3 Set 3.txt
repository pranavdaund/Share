1.  Display patient name, doctor name, and department for completed appointments.

		select p.patientname, d.doctorname,dpt.DepartmentName 
		from appointments app
		join patients p on p.Patientid = app.patientid
		join doctors d on d.doctorid = app.doctorid
		join departments dpt on d.DepartmentID = dpt.DepartmentID
		where Appointmentstatus = 'Completed';
		
		
2.  List appointments in March 2024 with treatment cost > 600.

		SELECT * FROM Appointments
		WHERE AppointmentDate >= '2024-03-01' 
		AND TreatmentCost > 600;
		
		
3.  Show doctors from Cardiology department who handled more than 3 appointments.
		 
	    SELECT d.DoctorName, COUNT(a.AppointmentID) AS TotalAppointments
		FROM Doctors d
		JOIN Departments dept ON d.DepartmentID = dept.DepartmentID
		JOIN Appointments a ON d.DoctorID = a.DoctorID
		WHERE dept.DepartmentName = 'Cardiology'
		GROUP BY d.DoctorName
		HAVING COUNT(a.AppointmentID) > 3;


4.  Display total revenue generated per department.
		
		SELECT  d.DepartmentName, SUM(a.TreatmentCost) AS TotalRevenue
		FROM Appointments a
		JOIN Doctors doc ON a.DoctorID = doc.DoctorID
		JOIN Departments d ON doc.DepartmentID = d.DepartmentID
		GROUP BY d.DepartmentName;
		
		
5.  List patients who never had a cancelled appointment.
      
		SELECT p.PatientName, a.AppointmentStatus
        FROM Patients p 
		JOIN Appointments a ON p.PatientID = a.PatientID
		WHERE a.AppointmentStatus = 'Cancelled';


6.  Show doctors earning more than 140000 who handled completed appointments.

		select DISTINCT d.doctorname 
		from doctors d
		join appointments a on d.doctorid = a.doctorID
		where d.salary > 140000 and a.appointmentstatus = 'Completed';
		

7.  Display patients treated by doctors from Oncology department.

		SELECT DISTINCT p.PatientName
		FROM Patients p
		JOIN Appointments a ON p.PatientID = a.PatientID
		JOIN Doctors d ON a.DoctorID = d.DoctorID
		JOIN Departments dep ON d.DepartmentID = dep.DepartmentID
		WHERE dep.DepartmentName = 'Oncology';
			

8.  Show total appointments per month.

		SELECT MONTH(AppointmentDate) AS Month, COUNT(*) AS TotalAppointments
		FROM Appointments
		GROUP BY MONTH(AppointmentDate)
		ORDER BY Month;

9.  Find top 3 doctors based on total treatment revenue generated.
		SELECT d.DoctorName, SUM(a.TreatmentCost) AS TotalRevenue
		FROM Doctors d
		JOIN Appointments a ON d.DoctorID = a.DoctorID
		WHERE a.AppointmentStatus = 'Completed'
		GROUP BY d.DoctorName
		ORDER BY TotalRevenue DESC
		LIMIT 3;


10. Display patients from USA who had appointments in April 2024.
		SELECT DISTINCT p.PatientName
		FROM Patients p
		JOIN Appointments a ON p.PatientID = a.PatientID
		WHERE p.Country = 'USA'
		AND a.AppointmentDate BETWEEN '2024-04-01' AND '2024-04-30';


11. List departments that had no cancelled appointments.

		SELECT dep.DepartmentName
		FROM Departments dep
		WHERE dep.DepartmentID NOT IN (
			SELECT DISTINCT d.DepartmentID
			FROM Doctors d
			JOIN Appointments a ON d.DoctorID = a.DoctorID
			WHERE a.AppointmentStatus = 'Cancelled'
		);
		
		
12. Show doctors who handled more than 5 completed appointments.

		SELECT d.DoctorName, COUNT(*) AS CompletedAppointments
		FROM Doctors d
		JOIN Appointments a ON d.DoctorID = a.DoctorID
		WHERE a.AppointmentStatus = 'Completed'
		GROUP BY d.DoctorName
		HAVING COUNT(*) > 5;


13. Display total revenue generated by each doctor.

		SELECT d.DoctorName, SUM(a.TreatmentCost) AS TotalRevenue
		FROM Doctors d
		JOIN Appointments a ON d.DoctorID = a.DoctorID
		WHERE a.AppointmentStatus = 'Completed'
		GROUP BY d.DoctorName;

14. List patients who visited more than 2 different departments.

		SELECT p.PatientName
		FROM Patients p
		JOIN Appointments a ON p.PatientID = a.PatientID
		JOIN Doctors d ON a.DoctorID = d.DoctorID
		GROUP BY p.PatientName
		HAVING COUNT(DISTINCT d.DepartmentID) > 2;


15. Show appointments where treatment cost is greater than average treatment cost.

		SELECT *
		FROM Appointments
		WHERE TreatmentCost > (
			SELECT AVG(TreatmentCost) FROM Appointments
		);


16. Display department-wise average treatment cost.

		SELECT dep.DepartmentName, AVG(a.TreatmentCost) AS AvgTreatmentCost
		FROM Appointments a
		JOIN Doctors d ON a.DoctorID = d.DoctorID
		JOIN Departments dep ON d.DepartmentID = dep.DepartmentID
		GROUP BY dep.DepartmentName;


17. List doctors who handled appointments in every month from Jan to June 2024.

		SELECT d.DoctorName
		FROM Doctors d
		JOIN Appointments a ON d.DoctorID = a.DoctorID
		WHERE a.AppointmentDate BETWEEN '2024-01-01' AND '2024-06-30'
		GROUP BY d.DoctorName
		HAVING COUNT(DISTINCT MONTH(a.AppointmentDate)) = 6;


18. Show patients who had both Scheduled and Completed appointments.

		SELECT p.PatientName
		FROM Patients p
		JOIN Appointments a ON p.PatientID = a.PatientID
		GROUP BY p.PatientName
		HAVING SUM(a.AppointmentStatus = 'Scheduled') > 0
		AND SUM(a.AppointmentStatus = 'Completed') > 0;


19. Display total completed appointments per department.

		SELECT dep.DepartmentName, COUNT(*) AS CompletedAppointments
		FROM Appointments a
		JOIN Doctors d ON a.DoctorID = d.DoctorID
		JOIN Departments dep ON d.DepartmentID = dep.DepartmentID
		WHERE a.AppointmentStatus = 'Completed'
		GROUP BY dep.DepartmentName;


20. Find doctors who never had a cancelled appointment.

		SELECT d.DoctorName
		FROM Doctors d
		WHERE d.DoctorID NOT IN (
			SELECT DoctorID
			FROM Appointments
			WHERE AppointmentStatus = 'Cancelled'
		);
